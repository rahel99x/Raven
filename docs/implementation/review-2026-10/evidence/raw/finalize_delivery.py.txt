"""Retain final documentation-containing distribution checks without rebinding regressions."""
from datetime import datetime, timezone
import gzip
import hashlib
import json
from pathlib import Path
import runpy
import shutil
import subprocess

ROOT = Path('/workspace/Raven')
BASE = ROOT / 'docs/implementation/review-2026-10'
NAME = 'py3.12-delivery'
OUTPUT = Path('/workspace/raven-qualification-results/final') / NAME
report = json.loads((OUTPUT / 'gates.json').read_text())
qualified = json.loads((BASE / 'evidence/final-qualification.json').read_text())
assert report['passed'] and report['source_stable'] and report['packaging_inputs_stable']
assert len(report['checks']) == 25 and all(c['exit_code'] == 0 for c in report['checks'])
assert report['source_before'] == report['source_after'] == qualified['source_fingerprint']
assert report['source_commit'] == subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT,text=True).strip()
assert not subprocess.check_output(['git','status','--porcelain=v1'],cwd=ROOT,text=True).strip()
for name, record in report['artifacts'].items():
    data = (OUTPUT/'dist'/name).read_bytes()
    assert len(data) == record['bytes'] and hashlib.sha256(data).hexdigest() == record['sha256']
retained = runpy.run_path('/workspace/raven-qualification-results/collect_final_evidence.py')['retain'](NAME)
for item in retained:
    data = (BASE/'evidence'/item['retained_path']).read_bytes()
    assert hashlib.sha256(data).hexdigest() == item['retained_sha256']
    original = gzip.decompress(data) if item['gzip'] else data
    assert len(original) == item['original_bytes'] and hashlib.sha256(original).hexdigest() == item['original_sha256']
receipt = {
    'schema':'raven.delivery-packaging/v1',
    'recorded_utc':datetime.now(timezone.utc).isoformat(),
    'status':'passed',
    'documentation_build_commit':report['source_commit'],
    'implementation_commit':qualified['implementation_commit'],
    'qualified_source_fingerprint':report['source_after'],
    'python':report['interpreter'],
    'checks':len(report['checks']),
    'gate_report':'raw/final-ci/'+NAME+'/gates.json',
    'artifacts':report['artifacts'],
    'artifact_workspace_directory':str(OUTPUT/'dist'),
    'archives':retained,
    'scope':'Sequential clean wheel/sdist, extracted-source, schemas, references, installed-command, terminal, flagship and inventory gates on the committed complete handback. The five-interpreter regressions remain bound to e3c7183; runtime/tests/scripts/schemas/metadata/CI are unchanged. This outer delivery receipt and evidence-index refresh are added after building and do not claim to be inside their own artifacts.',
    'limits':['Local verification only; no hosted CI, publication, signature, push or merge.','The binary artifacts reside in the workspace; exact digests, sizes, inventories and lossless logs are retained in the repository.']
}
(BASE/'evidence/delivery-packaging.json').write_text(json.dumps(receipt,indent=2)+'\n')
readme=BASE/'README.md'
text=readme.read_text()
needle='The original Python 3.10 API failure, Python 3.13 watch-timeout failure and\n'
assert needle in text
addition=('A final [delivery packaging receipt](evidence/delivery-packaging.json) records\n'
          '25 additional passing checks on distributions built from the completed\n'
          'documentation commit `'+report['source_commit'][:7]+'`. The five-interpreter regression binding\n'
          'above remains unchanged. This outer receipt was added after the build.\n\n')
readme.write_text(text.replace(needle,addition+needle,1))
p=BASE/'progress.md'
p.write_text(p.read_text()+'\n## Final delivery packaging\n\nAll 25 sequential Python 3.12 delivery gates passed on documentation commit `'+report['source_commit']+'`. The completed handback was included in the wheel/source qualification; the exact artifacts, inventories and lossless logs are bound in [the delivery receipt](evidence/delivery-packaging.json). Runtime, tests, scripts, schemas, packaging metadata and CI remain identical to the final five-interpreter-qualified revision. The receipt is outer verification metadata, added after building; no hosted run or publication is implied.\n')
shutil.copyfile('/workspace/raven-qualification-results/finalize_delivery.py',BASE/'evidence/raw/finalize_delivery.py.txt')
print(json.dumps({'status':'passed','checks':25,'retained_files':len(retained),'build_commit':report['source_commit'],'artifacts':report['artifacts']},indent=2))
