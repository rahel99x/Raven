"""Close only completed local gates; preserve earlier source-bound status evidence."""
from __future__ import annotations
from collections import Counter
import hashlib
import json
from pathlib import Path
import re
import subprocess

ROOT = Path('/workspace/Raven')
BASE = ROOT / 'docs/implementation/review-2026-10'
EVIDENCE = BASE / 'evidence'
IMPLEMENTATION = 'e3c7183286fccdda68660b5541ec2e8c6d2c7750'
RUNTIME = '0d138fdacbcfb2d58c3d77a0494d439c7d2204e8'
EXTERNAL = {'W09-R04', 'W19-R06', 'W32-R09', 'W39-R09', 'W42-R02',
            'W44-R03', 'W44-R06', 'W44-R08'}
GATE = 'Local integrated gates passed; see evidence/final-qualification.json. Hosted CI is not claimed.'
REPLACEMENTS = {
    'Focused working-tree checks only; final integrated revision checks still required.': GATE,
    'Final integrated revision/CI validation pending; package checks bind the current source hashes.': GATE + ' Earlier focused checks retain their original source bindings.',
    'Final source-frozen controlled performance rerun and required integrated checks remain release gates; initial measured correctness overhead is recorded, not presented as speedup.': 'The controlled state rerun is retained in evidence/state-before-final.json and evidence/state-after-final.json; correctness overhead is recorded, not presented as speedup. ' + GATE,
    'Final integrated revision, packaging, installed commands and CI remain root-owned release gates.': GATE,
    'Final integrated revision regression/packaging pending.': GATE,
    'Final integrated revision validation remains pending.': GATE,
    'Final integrated revision validation pending.': GATE,
    'Final integrated revision checks pending.': GATE,
    'Final integrated checks pending.': GATE,
    'final root integrated revision checks pending.': GATE,
    'final integrated revision rerun pending root matrix.': GATE,
    'Human terminal/remote UX and final integrated revision gates remain separate.': 'Human terminal/remote UX remains separately unqualified. ' + GATE,
}


def write(path, value):
    path.write_text(json.dumps(value, indent=2, ensure_ascii=False) + '\n')


def reconcile_text(value):
    if isinstance(value, str):
        for old, new in REPLACEMENTS.items():
            value = value.replace(old, new)
        return value
    if isinstance(value, list):
        return [reconcile_text(item) for item in value]
    if isinstance(value, dict):
        return {key: reconcile_text(item) for key, item in value.items()}
    return value


def main():
    final = json.loads((EVIDENCE / 'final-qualification.json').read_text())
    assert final['implementation_commit'] == IMPLEMENTATION
    assert len(final['matrix']) == 5
    assert all(row['packaging_checks'] == 25 and row['counts']['passed'] > 18000
               for row in final['matrix'])
    originals = {path.name: json.loads(path.read_text())
                 for path in sorted(EVIDENCE.glob('W*-status.json'))}
    assert len(originals) == 44
    expected_counts = Counter()
    expected_blocked = set()
    for original in originals.values():
        rs = original['requirements']
        if isinstance(rs, dict):
            rs = [dict(value, id=key) if isinstance(value, dict) else {'id': key, 'state': value}
                  for key, value in rs.items()]
        for req in rs:
            disposition = 'implemented' if req['id'] == 'W22-R05' else req.get('status', req.get('state'))
            expected_counts[disposition] += 1
            if disposition == 'blocked':
                expected_blocked.add(req['id'])
            refs = req.get('evidence', [])
            refs = [refs] if isinstance(refs, str) else refs
            for ref in refs:
                parts = ref.partition(' — ')[0].split(' and ')
                for part in parts:
                    if '/' not in part and len(parts) > 1:
                        part = str(Path(parts[0]).parent / part)
                    file_part = part.partition('#')[0]
                    assert (BASE / file_part).is_file() or (ROOT / file_part).is_file(), (req['id'], part)
    assert expected_blocked == EXTERNAL, expected_blocked
    assert expected_counts == {'implemented': 312, 'already_satisfied': 35, 'adjusted': 16, 'blocked': 8}, expected_counts
    historical = EVIDENCE / 'raw/pre-final-package-statuses.json'
    assert not historical.exists(), 'Finalizer must not silently overwrite history'
    write(historical, {'purpose': 'Focused package statuses before final integrated gate closure; stale pending language is historical.',
                       'statuses': originals})
    integrated = {'implementation_revision': IMPLEMENTATION, 'runtime_revision': RUNTIME,
                  'source_fingerprint': final['source_fingerprint'],
                  'evidence': 'evidence/final-qualification.json',
                  'outcome': 'Supported Python 3.10–3.14 full regression and 25 packaging/installed/schema/reference checks per interpreter passed locally.',
                  'limits': 'Optional tool skips, actual supplementary qualifications, experimental milestones and external gates remain distinct. No hosted CI or publication is claimed.'}
    for name, original in originals.items():
        status = reconcile_text(original)
        for field in ('candidate', 'candidate_revision', 'validation', 'qualification'):
            if field in status:
                status.setdefault('focused_history', {})[field] = original[field]
                del status[field]
        status['tested_revision'] = IMPLEMENTATION
        status['integrated_validation'] = integrated
        status['validation'] = integrated['outcome'] + ' Historical focused results and measurements retain their original revision/hash bindings.'
        status['historical_status'] = 'evidence/raw/pre-final-package-statuses.json'
        requirements = status['requirements']
        if isinstance(requirements, dict):
            requirements = [dict(value, id=key) if isinstance(value, dict)
                            else {'id': key, 'state': value} for key, value in requirements.items()]
            status['requirements'] = requirements
        for req in requirements:
            req['state'] = req.pop('status', req.get('state'))
            if req['id'] == 'W22-R05':
                req['state'] = 'implemented'
                req.pop('blocker_type', None)
                req.pop('unblock_condition', None)
                req['conclusion'] = 'Final wheel/sdist builds, extracted-source fixture tests, clean optional-free installed commands, helper shipping, schema/reference/quality checks and supported Python matrix passed. Exact commands and artifacts are retained.'
            evidence = req.get('evidence', [])
            if isinstance(evidence, str):
                evidence = [evidence]
            paths = []
            for item in evidence:
                # Preserve prose as a note; make every evidence entry an exact path.
                path_part, _, note = item.partition(' — ')
                parts = path_part.split(' and ')
                parent = Path(parts[0]).parent
                for part in parts:
                    if '/' not in part and len(parts) > 1:
                        part = str(parent / part)
                    file_part = part.partition('#')[0]
                    assert (BASE / file_part).is_file() or (ROOT / file_part).is_file(), (req['id'], part)
                    if part not in paths:
                        paths.append(part)
                if note:
                    req.setdefault('evidence_notes', []).append(note)
            if 'evidence/final-qualification.json' not in paths:
                paths.append('evidence/final-qualification.json')
            req['evidence'] = paths
        blocked = [req['id'] for req in requirements if req['state'] == 'blocked']
        status['state'] = 'locally_validated_with_external_gates' if blocked else 'locally_validated'
        status['external_gates'] = blocked
        write(EVIDENCE / name, status)
        md = EVIDENCE / (status['package'] + '.md')
        md.write_text(md.read_text() + '\n\n## Final local integration\n\n'
                      f'Implementation `{IMPLEMENTATION}` (runtime `{RUNTIME}`) passed the '
                      '[final Python 3.10–3.14 regression and packaging gates](final-qualification.json). '
                      'This supersedes earlier pending-integration language in this chronological record. '
                      'Earlier measurements and focused runs keep their recorded revisions and source hashes. '
                      'The [current package status](' + name + ') retains experimental, platform, hardware, '
                      'human-study and deployment limits; local tests do not close external qualification.\n')
    subprocess.run([str(ROOT / '.venv/bin/python'), str(BASE / 'refresh-coverage.py')], check=True, cwd=ROOT)
    ledger_path = BASE / 'coverage.json'
    ledger = json.loads(ledger_path.read_text())
    counts = Counter()
    blocked_ids = set()
    for package in ledger['packages']:
        status = json.loads((EVIDENCE / (package['id'] + '-status.json')).read_text())
        package['tested_revision'] = IMPLEMENTATION
        package['state'] = status['state']
        package['verification'] = 'local_integrated_gates_passed'
        package['integrated_validation'] = integrated
        package['review_evidence'] = [package['evidence_file'], 'evidence/maintainer-review.json']
        package['validation_summary'] = status['validation']
        package['next_action'] = 'Satisfy explicit external gates before production qualification.' if status['external_gates'] else 'Review local changes; preserve documented feature qualification limits.'
        for req in package['requirements']:
            counts[req['state']] += 1
            if req['state'] == 'blocked':
                blocked_ids.add(req['id'])
    assert blocked_ids == EXTERNAL, blocked_ids
    assert counts == {'implemented': 312, 'already_satisfied': 35, 'adjusted': 16, 'blocked': 8}, counts
    ledger['implementation_revision'] = IMPLEMENTATION
    ledger['runtime_revision'] = RUNTIME
    ledger['integrated_validation'] = integrated
    ledger['evidence_path_roots'] = 'evidence/ and measurements/ resolve relative to this ledger; raven/, tests/, scripts/ and docs/ resolve relative to the repository root. Fragments identify Markdown headings.'
    ledger['dispositions'] = dict(counts)
    ledger['external_gates'] = sorted(EXTERNAL)
    ledger['evidence_note'] = 'All 371 exact clauses reconciled. Local integrated gates passed with recorded optional skips; eight external clauses remain blocked with evidence and unblock conditions. Measurements retain original source bindings. No hosted CI, release, PR, push or merge is claimed.'
    write(ledger_path, ledger)
    print(json.dumps({'dispositions': dict(counts), 'external_gates': sorted(EXTERNAL)}, indent=2))


if __name__ == '__main__':
    main()
